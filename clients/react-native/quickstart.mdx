**Expected time:** 15 minutes after the version-locked source checkouts,
gateway resources, Firebase identity, and one native host configuration exist;
repeat the native run on the other platform for completion proof.

**Mode:** Source evaluation; iOS and Android compilation is distinct from
protected App Attest, Play Integrity, or distribution evidence.

**Coordinates:** server contract `1.0.0`; React Native SDK source `1.0.0` with
its locked iOS and Android SDK checkpoints.

## Outcome

Run one complete TypeScript screen on iOS and Android, bind its fetch function
to `habit-assistant`, consume a streamed Responses body, and find both requests
in the Console. Native modules continue to own component keys, DPoP, session
rotation, and platform attestation.

## Before you start

- a reachable Latchway `1.0.0` gateway with an application resource ID,
  environment, `habit-assistant` feature, Responses route, and limits
- authorized, version-locked checkouts of the React Native, iOS, and Android
  SDK source candidates
- React Native 0.82 with the New Architecture enabled
- a signed-in Firebase user whose ID token the gateway identity provider accepts
- either the iOS 15+ toolchain or the Android API 24+ toolchain for the first
  run; both are required for the final React Native completion proof

<Note>
  The Firebase packages belong to the checked-in example, not the Latchway
  runtime package. You can use another supported identity provider by replacing
  only `getIdentityToken`; it must return your application's current user JWT.
  Never put an upstream provider key in React Native code or configuration.
</Note>

## What you will build

```text
TypeScript feature-bound fetch
  -> native iOS or Android module
  -> platform key + native DPoP signing
  -> Latchway feature and server-owned route
```

JavaScript never receives a private key, attestation evidence, access token, or
refresh token from the native bridge.

```tsx
import { useEffect, useRef, useState } from "react";
import {
  Button,
  Linking,
  Platform,
  SafeAreaView,
  ScrollView,
  Text,
  TextInput,
  View,
} from "react-native";
import firebaseAuth from "@react-native-firebase/auth";
import {
  createLatchwayClient,
  errorFromResponse,
  LatchwayError,
  type LatchwayClient,
} from "@latchway/react-native";

const configuration = {
  baseURL: "https://gateway.example.com",
  applicationID: "app_01J00000000000000000000000",
  environment: "development",
  feature: "habit-assistant",
  model: "latchway",
  iosRootKeychainAccessGroup: "ABCDE12345.com.example.habits",
  androidPlayIntegrityCloudProjectNumber: "123456789012",
} as const;

async function getIdentityToken(): Promise<string> {
  const user = firebaseAuth().currentUser;
  if (user === null) {
    throw new Error("Sign in with Firebase before calling Latchway.");
  }
  return user.getIdToken();
}

function createAppClient(): LatchwayClient {
  const shared = {
    baseURL: configuration.baseURL,
    applicationID: configuration.applicationID,
    environment: configuration.environment,
    identityProvider: "firebase",
    getIdentityToken,
  };

  if (Platform.OS === "ios") {
    return createLatchwayClient({
      ...shared,
      apple: {
        rootKeychainAccessGroup: configuration.iosRootKeychainAccessGroup,
      },
    });
  }
  if (Platform.OS === "android") {
    return createLatchwayClient({
      ...shared,
      android: {
        playIntegrityCloudProjectNumber:
          configuration.androidPlayIntegrityCloudProjectNumber,
      },
    });
  }
  throw new Error("This example supports only native iOS and Android hosts.");
}

const latchway = createAppClient();
const habitAssistantFetch = latchway.fetchFor(configuration.feature);

export default function App(): React.JSX.Element {
  const [input, setInput] = useState("Plan a focused afternoon.");
  const [output, setOutput] = useState("");
  const [status, setStatus] = useState("Ready");
  const [documentationURL, setDocumentationURL] = useState<string>();
  const [busy, setBusy] = useState(false);
  const activeRequest = useRef<AbortController | null>(null);

  useEffect(() => () => {
    activeRequest.current?.abort();
    void latchway.dispose();
  }, []);

  const send = async (): Promise<void> => {
    if (activeRequest.current !== null) return;
    const controller = new AbortController();
    activeRequest.current = controller;
    setBusy(true);
    setOutput("");
    setDocumentationURL(undefined);
    try {
      await latchway.ready;
      const response = await habitAssistantFetch("/v1/responses", {
        method: "POST",
        signal: controller.signal,
        headers: {
          Accept: "text/event-stream",
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          model: configuration.model,
          input,
          stream: true,
        }),
      });
      if (!response.ok) throw await errorFromResponse(response);

      const requestID = safeRequestID(
        response.headers.get("X-Latchway-Request-ID"),
      );
      setStatus(requestID === undefined
        ? "Streaming response"
        : `Streaming · request ${requestID}`);
      await consumeBoundedStream(response, (text) => {
        setOutput((current) => current + text);
      });
      setStatus(requestID === undefined
        ? "Completed"
        : `Completed · request ${requestID}`);
    } catch (error) {
      if (error instanceof Error && error.name === "AbortError") {
        setStatus("Cancelled; start a new request when ready.");
      } else if (error instanceof LatchwayError) {
        const request = error.requestID === undefined
          ? ""
          : ` · request ${error.requestID}`;
        setStatus(`${error.code}${request}: ${error.message}`);
        setDocumentationURL(error.documentationURL);
      } else {
        setStatus(error instanceof Error ? error.message : "Request failed.");
      }
    } finally {
      if (activeRequest.current === controller) activeRequest.current = null;
      setBusy(false);
    }
  };

  return (
    <SafeAreaView style={{ flex: 1 }}>
      <ScrollView contentContainerStyle={{ gap: 12, padding: 24 }}>
        <Text>Habit assistant</Text>
        <TextInput
          editable={!busy}
          multiline
          onChangeText={setInput}
          value={input}
        />
        <View style={{ gap: 8 }}>
          <Button
            disabled={busy || input.trim().length === 0}
            onPress={() => { void send(); }}
            title={busy ? "Streaming…" : "Send request"}
          />
          <Button
            disabled={!busy}
            onPress={() => { activeRequest.current?.abort(); }}
            title="Cancel"
          />
        </View>
        <Text selectable>{status}</Text>
        <Text selectable>{output}</Text>
        {documentationURL === undefined ? null : (
          <Button
            onPress={() => { void Linking.openURL(documentationURL); }}
            title="Open error guide"
          />
        )}
      </ScrollView>
    </SafeAreaView>
  );
}

async function consumeBoundedStream(
  response: Response,
  append: (text: string) => void,
): Promise<void> {
  if (response.body === null) {
    throw new Error("Gateway did not return a response stream.");
  }
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let bytes = 0;
  try {
    for (;;) {
      const chunk = await reader.read();
      if (chunk.done) break;
      bytes += chunk.value.byteLength;
      if (bytes > 1_048_576) {
        await reader.cancel("Quickstart display limit reached");
        throw new Error("Stream exceeded the 1 MiB display limit.");
      }
      append(decoder.decode(chunk.value, { stream: true }));
    }
    const tail = decoder.decode();
    if (tail.length > 0) append(tail);
  } finally {
    reader.releaseLock();
  }
}

function safeRequestID(value: string | null): string | undefined {
  return value !== null && /^[A-Za-z0-9][A-Za-z0-9._:-]{7,127}$/u.test(value)
    ? value
    : undefined;
}
```

## Troubleshooting

<AccordionGroup>
  <Accordion title="The native module is unavailable">
    Confirm React Native 0.82 and the New Architecture, then rebuild the native
    host; a Metro reload cannot add a TurboModule. On iOS, confirm both
    Latchway pods in `Podfile.lock`. On Android, confirm the `dev.latchway`
    dependency insight. Continue with [iOS native setup](/clients/react-native/ios-native-setup)
    or [Android native setup](/clients/react-native/android-native-setup).
  </Accordion>
  <Accordion title="The request returns a typed Latchway error">
    Use **Open error guide** for the exact stable code, then correlate its safe
    request ID in the Console. Check the gateway origin, generated application
    resource ID, environment, feature grant, and signed-in identity. Respect the
    error's retry guidance; never blindly replay a consumed stream or an
    indeterminate dispatch. See [React Native error handling](/clients/react-native/error-handling).
  </Accordion>
  <Accordion title="Attestation is unsupported or required">
    Simulator, emulator, sideloaded, and Debug results do not establish the
    protected platform verdict. Check the exact iOS entitlement and provisioned
    bundle ID, or the Android cloud project, package, certificate, Play track,
    and install source. Continue with [physical-device proof](/mobile/device-proof).
  </Accordion>
</AccordionGroup>
